# ==========================================
# 2. ロジック部分
# ==========================================

colors = ["blue", "red", "yellow", "green", "pink", "black", "white", "gold", "tiger pattern", "silver", "neon orange", "metallic purple"]
locations = ["park", "office", "train", "beach", "onsen", "space", "home", "bar", "gym", "mountain", "sauna", "construction site"]

def generate_csv_data():
    # 6つのテーマ（色×場所）をランダム生成
    all_combinations = [(c, l) for c in colors for l in locations]
    selected_themes = random.sample(all_combinations, 6)

    # 6つのテキストパターンをランダムに割り当て
    # Standardが4つあるので、高確率でStandard系が選ばれます
    selected_pattern_names = random.sample(list(all_patterns.keys()), 6)

    print("--- Copy below and paste into Variable column ---")

    for i, (color, location) in enumerate(selected_themes):
        # このテーマに使うテキストリストを取得
        pattern_name = selected_pattern_names[i]
        text_list = all_patterns[pattern_name]
        
        # コンソールに、どのパターンが選ばれたか表示（デバッグ用）
        # print(f"# Theme {i+1}: {color} / {location} -> {pattern_name}")
        
        # 45個のワードを、9個ずつ5回に分けて使う
        for j in range(5):
            start_index = j * 9
            end_index = start_index + 9
            
            if end_index > len(text_list):
                current_texts = text_list[:9]
            else:
                current_texts = text_list[start_index:end_index]
            
            text_prompt_part = ", ".join([f"'{t}'" for t in current_texts])
            
            prompt = (
                f"{color} zentai full body suit, {location}, "
                f"various poses, "
                f"including speech bubbles with Japanese text captions chosen from: {text_prompt_part}"
            )
            
            print(prompt)

if __name__ == "__main__":
    generate_csv_data()
